class Solution {
    public int sumOfTheDigitsOfHarshadNumber(int x) {
        int sum = 0;
        int r, y = x;

        while (y > 0) {
            r = y % 10;
            sum += r;
            y = y / 10;
        }

        if (x % sum == 0)
            return sum;
        else
            return -1;
    }
}
