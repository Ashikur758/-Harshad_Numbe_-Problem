import java.util.*;
import java.io.*;

public class Main {
    public static void main(String[] args) throws IOException {
        int MAX = 1000000;
        boolean[] isNotDevlali = new boolean[MAX + 1];
        boolean[] isPrime = new boolean[MAX + 1];
        
        // 1. Sieve of Eratosthenes for Primes
        Arrays.fill(isPrime, true);
        isPrime[0] = isPrime[1] = false;
        for (int p = 2; p * p <= MAX; p++) {
            if (isPrime[p]) {
                for (int i = p * p; i <= MAX; i += p)
                    isPrime[i] = false;
            }
        }

        // 2. Identify Devlali Numbers
        // A number is NOT Devlali if it can be generated by d(r)
        for (int r = 1; r <= MAX; r++) {
            int dr = d(r);
            if (dr <= MAX) {
                isNotDevlali[dr] = true;
            }
        }

        // 3. Precompute Prefix Sums of Devlali Primes
        int[] prefixSum = new int[MAX + 1];
        int currentCount = 0;
        for (int i = 1; i <= MAX; i++) {
            if (!isNotDevlali[i] && isPrime[i]) {
                currentCount++;
            }
            prefixSum[i] = currentCount;
        }

        // 4. Fast I/O for Queries
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null) return;
        
        int Q = Integer.parseInt(line.trim());
        StringBuilder out = new StringBuilder();

        while (Q-- > 0) {
            String[] range = br.readLine().split("\\s+");
            int A = Integer.parseInt(range[0]);
            int B = Integer.parseInt(range[1]);

            // Handle range boundaries
            if (A < 1) A = 1;
            if (B < 1) {
                out.append("0\n");
            } else {
                int result = prefixSum[B] - prefixSum[A - 1];
                out.append(result).append("\n");
            }
        }
        System.out.print(out);
    }

    // Helper function to calculate d(n)
    private static int d(int n) {
        int sum = n;
        while (n > 0) {
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
}
